-- 1. TẮT KIỂM TRA KHÓA NGOẠI ĐỂ DỄ DÀNG XÓA BẢNG CŨ
SET FOREIGN_KEY_CHECKS = 0;

CREATE DATABASE IF NOT EXISTS flower_shop CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE flower_shop;

-- 2. XÓA CÁC BẢNG CŨ NẾU ĐÃ TỒN TẠI
DROP TABLE IF EXISTS careschedule;
DROP TABLE IF EXISTS payment;
DROP TABLE IF EXISTS orderdetail;
DROP TABLE IF EXISTS `order`;
DROP TABLE IF EXISTS flowerbatch;
DROP TABLE IF EXISTS purchaseorder;
DROP TABLE IF EXISTS flower;
DROP TABLE IF EXISTS supplier;
DROP TABLE IF EXISTS employee;
DROP TABLE IF EXISTS customer;

-- =====================================================
-- 3. TẠO CÁC BẢNG (CHỮ THƯỜNG 100%)
-- =====================================================

CREATE TABLE customer (
    customer_id VARCHAR(10) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    phone VARCHAR(15) NOT NULL UNIQUE, 
    address VARCHAR(255),
    email VARCHAR(100)
);

CREATE TABLE employee (
    employee_id VARCHAR(10) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    role VARCHAR(50),
    phone VARCHAR(15),
    salary DECIMAL(15, 2)
);

CREATE TABLE supplier (
    supplier_id VARCHAR(10) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    phone VARCHAR(15) NOT NULL UNIQUE,
    address VARCHAR(255),
    representative VARCHAR(100)
);

CREATE TABLE flower (
    flower_id VARCHAR(10) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    category VARCHAR(50),
    unit_price DECIMAL(15, 2),
    stock_quantity INT DEFAULT 0,
    color VARCHAR(30),
    supplier_id VARCHAR(10),
    FOREIGN KEY (supplier_id) REFERENCES supplier(supplier_id) ON DELETE SET NULL
);

CREATE TABLE purchaseorder (
    purchase_order_id VARCHAR(10) PRIMARY KEY,
    supplier_id VARCHAR(10),
    employee_id VARCHAR(10),
    import_date DATETIME,
    total_amount DECIMAL(15, 2),
    status VARCHAR(50),
    note TEXT,
    FOREIGN KEY (supplier_id) REFERENCES supplier(supplier_id),
    FOREIGN KEY (employee_id) REFERENCES employee(employee_id)
);

CREATE TABLE flowerbatch (
    batch_id VARCHAR(10) PRIMARY KEY,
    purchase_order_id VARCHAR(10),
    flower_id VARCHAR(10),
    quantity_imported INT,
    quantity_remaining INT,
    unit_cost DECIMAL(15, 2),
    import_date DATETIME,
    expiry_date DATE,
    status VARCHAR(50),
    note TEXT,
    FOREIGN KEY (purchase_order_id) REFERENCES purchaseorder(purchase_order_id),
    FOREIGN KEY (flower_id) REFERENCES flower(flower_id)
);

-- Lưu ý: order là từ khóa nên phải bọc bằng dấu huyền `order`
CREATE TABLE `order` (
    order_id VARCHAR(10) PRIMARY KEY,
    customer_id VARCHAR(10),
    employee_id VARCHAR(10),
    order_date DATETIME,
    delivery_date DATETIME,
    delivery_address VARCHAR(255),
    order_status VARCHAR(50),
    FOREIGN KEY (customer_id) REFERENCES customer(customer_id),
    FOREIGN KEY (employee_id) REFERENCES employee(employee_id)
);

CREATE TABLE orderdetail (
    order_id VARCHAR(10),
    flower_id VARCHAR(10),
    quantity INT,
    unit_price_at_sale DECIMAL(15, 2),
    PRIMARY KEY (order_id, flower_id),
    FOREIGN KEY (order_id) REFERENCES `order`(order_id) ON DELETE CASCADE,
    FOREIGN KEY (flower_id) REFERENCES flower(flower_id)
);

CREATE TABLE payment (
    payment_id VARCHAR(10) PRIMARY KEY,
    order_id VARCHAR(10),
    amount DECIMAL(15, 2),
    payment_method VARCHAR(50),
    payment_date DATETIME,
    payment_status VARCHAR(50),
    FOREIGN KEY (order_id) REFERENCES `order`(order_id)
);

-- Bảng chăm sóc liên kết với MÃ LÔ (batch_id) thay vì Mã Hoa
CREATE TABLE careschedule (
    care_id VARCHAR(10) PRIMARY KEY,
    batch_id VARCHAR(10),
    employee_id VARCHAR(10),
    check_date DATETIME,
    freshness_level VARCHAR(50),
    water_status VARCHAR(50),
    action_taken VARCHAR(255),
    note TEXT,
    FOREIGN KEY (batch_id) REFERENCES flowerbatch(batch_id),
    FOREIGN KEY (employee_id) REFERENCES employee(employee_id)
);
