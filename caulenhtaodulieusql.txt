DROP DATABASE IF EXISTS flower_shop;
CREATE DATABASE flower_shop CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE flower_shop;

-- =====================================================================
-- PHẦN 1: TẠO BẢNG
-- =====================================================================

CREATE TABLE Customer (
    customer_id VARCHAR(10) PRIMARY KEY,
    name        NVARCHAR(100) NOT NULL,
    phone       VARCHAR(15),
    address     NVARCHAR(255),
    email       VARCHAR(100)
);

CREATE TABLE Supplier (
    supplier_id    VARCHAR(10) PRIMARY KEY,
    name           NVARCHAR(150) NOT NULL,
    phone          VARCHAR(15),
    address        NVARCHAR(255),
    representative NVARCHAR(100)
);

CREATE TABLE Employee (
    employee_id VARCHAR(10) PRIMARY KEY,
    name        NVARCHAR(100) NOT NULL,
    role        NVARCHAR(50),
    phone       VARCHAR(15),
    salary      DECIMAL(12,2)
);

CREATE TABLE Flower (
    flower_id      VARCHAR(10) PRIMARY KEY,
    name           NVARCHAR(150) NOT NULL,
    category       NVARCHAR(50),
    unit_price     DECIMAL(12,2),
    stock_quantity INT DEFAULT 0,
    color          NVARCHAR(30),
    supplier_id    VARCHAR(10),
    FOREIGN KEY (supplier_id) REFERENCES Supplier(supplier_id)
);

CREATE TABLE `Order` (
    order_id         VARCHAR(10) PRIMARY KEY,
    customer_id      VARCHAR(10) NOT NULL,
    employee_id      VARCHAR(10) NOT NULL,
    order_date       DATETIME,
    delivery_date    DATETIME,
    delivery_address NVARCHAR(255),
    order_status     VARCHAR(30),
    FOREIGN KEY (customer_id) REFERENCES Customer(customer_id),
    FOREIGN KEY (employee_id) REFERENCES Employee(employee_id)
);

CREATE TABLE OrderDetail (
    order_id           VARCHAR(10),
    flower_id          VARCHAR(10),
    quantity           INT NOT NULL CHECK (quantity > 0),
    unit_price_at_sale DECIMAL(12,2) NOT NULL,
    PRIMARY KEY (order_id, flower_id),
    FOREIGN KEY (order_id)  REFERENCES `Order`(order_id) ON DELETE CASCADE,
    FOREIGN KEY (flower_id) REFERENCES Flower(flower_id)
);

CREATE TABLE Payment (
    payment_id     VARCHAR(10) PRIMARY KEY,
    order_id       VARCHAR(10) NOT NULL,
    amount         DECIMAL(12,2),
    payment_method VARCHAR(30),
    payment_date   DATETIME,
    payment_status VARCHAR(30),
    FOREIGN KEY (order_id) REFERENCES `Order`(order_id)
);

CREATE TABLE CareSchedule (
    care_id         VARCHAR(10) PRIMARY KEY,
    flower_id       VARCHAR(10) NOT NULL,
    employee_id     VARCHAR(10) NOT NULL,
    check_date      DATETIME,
    freshness_level NVARCHAR(50),
    water_status    NVARCHAR(50),
    action_taken    NVARCHAR(255),
    note            NVARCHAR(255),
    FOREIGN KEY (flower_id)   REFERENCES Flower(flower_id),
    FOREIGN KEY (employee_id) REFERENCES Employee(employee_id)
);

CREATE TABLE PurchaseOrder (
    purchase_order_id VARCHAR(10) PRIMARY KEY,
    supplier_id       VARCHAR(10) NOT NULL,
    employee_id       VARCHAR(10) NOT NULL,
    import_date       DATETIME,
    total_amount      DECIMAL(12,2),
    status            VARCHAR(30),
    note              NVARCHAR(255),
    FOREIGN KEY (supplier_id) REFERENCES Supplier(supplier_id),
    FOREIGN KEY (employee_id) REFERENCES Employee(employee_id)
);

CREATE TABLE FlowerBatch (
    batch_id           VARCHAR(10) PRIMARY KEY,
    purchase_order_id  VARCHAR(10) NOT NULL,
    flower_id          VARCHAR(10) NOT NULL,
    quantity_imported  INT,
    quantity_remaining INT,
    unit_cost          DECIMAL(12,2),
    import_date        DATETIME,
    expiry_date        DATE,
    status             VARCHAR(30),
    note               NVARCHAR(255),
    FOREIGN KEY (purchase_order_id) REFERENCES PurchaseOrder(purchase_order_id),
    FOREIGN KEY (flower_id)         REFERENCES Flower(flower_id)
);
