-- =====================================================
-- FLOWERSHOP V2 - SCRIPT TẠO CSDL ĐẦY ĐỦ (copy & paste)
-- MySQL 8 / MariaDB 10.2+  |  InnoDB  |  utf8mb4
-- (CHECK chỉ thực thi từ MySQL 8.0.16+ / MariaDB 10.2+; bản cũ sẽ bỏ qua nhưng vẫn chạy được)
--
-- Quy ước nghiệp vụ đang dùng trong code:
--   order_status    : DANG_CHUAN_BI, DANG_GIAO, HOAN_THANH, DA_HUY, GIAO_THAT_BAI
--                     (CHO_XU_LY / CHO_XAC_NHAN là trạng thái cũ, chỉ để tương thích)
--   payment_status  : THANH_TOAN_THANH_CONG, DA_HUY, DA_HOAN_TIEN
--   batch status    : AVAILABLE, SOLD_OUT, EXPIRED
--   PO status       : CHO_XU_LY, HOAN_THANH
--
-- Bất biến quan trọng (code dựa vào):
--   flower.stock_quantity = TỔNG quantity_remaining của các lô AVAILABLE
--   order.total không lưu; tổng đơn = SUM(orderdetail.quantity * unit_price_at_sale)
--   purchaseorder.total_amount = SUM(flowerbatch.quantity_imported * unit_cost)
-- =====================================================

SET FOREIGN_KEY_CHECKS = 0;

DROP DATABASE IF EXISTS flower_shop;
CREATE DATABASE flower_shop CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE flower_shop;

-- =====================================================
-- 1. KHÁCH HÀNG
-- =====================================================
CREATE TABLE customer (
    customer_id VARCHAR(10) PRIMARY KEY,
    name        VARCHAR(100) NOT NULL,
    phone       VARCHAR(15)  NOT NULL UNIQUE,
    address     VARCHAR(255),
    email       VARCHAR(100),
    CONSTRAINT ck_customer_name  CHECK (TRIM(name) <> ''),
    CONSTRAINT ck_customer_phone CHECK (phone REGEXP '^0[0-9]{9}$'),
    CONSTRAINT ck_customer_email CHECK (email IS NULL OR email = '' OR email LIKE '%@%')
) ENGINE = InnoDB;

-- =====================================================
-- 2. NHÂN VIÊN
-- =====================================================
CREATE TABLE employee (
    employee_id VARCHAR(10) PRIMARY KEY,
    name        VARCHAR(100) NOT NULL,
    role        VARCHAR(50),
    phone       VARCHAR(15),
    salary      DECIMAL(15, 2),
    CONSTRAINT ck_employee_salary CHECK (salary IS NULL OR salary > 0)
) ENGINE = InnoDB;

-- =====================================================
-- 3. NHÀ CUNG CẤP
-- =====================================================
CREATE TABLE supplier (
    supplier_id    VARCHAR(10) PRIMARY KEY,
    name           VARCHAR(100) NOT NULL,
    phone          VARCHAR(15)  NOT NULL UNIQUE,
    address        VARCHAR(255),
    representative VARCHAR(100),
    CONSTRAINT ck_supplier_name  CHECK (TRIM(name) <> ''),
    CONSTRAINT ck_supplier_phone CHECK (phone REGEXP '^0[0-9]{9,10}$')
) ENGINE = InnoDB;

-- =====================================================
-- 4. HOA
-- =====================================================
CREATE TABLE flower (
    flower_id      VARCHAR(10) PRIMARY KEY,
    name           VARCHAR(100) NOT NULL,
    category       VARCHAR(50),
    unit_price     DECIMAL(15, 2),
    stock_quantity INT DEFAULT 0,
    color          VARCHAR(30),
    supplier_id    VARCHAR(10),
    CONSTRAINT fk_flower_supplier FOREIGN KEY (supplier_id)
        REFERENCES supplier (supplier_id) ON DELETE RESTRICT,
    CONSTRAINT ck_flower_name  CHECK (TRIM(name) <> ''),
    CONSTRAINT ck_flower_price CHECK (unit_price IS NULL OR unit_price > 0),
    CONSTRAINT ck_flower_stock CHECK (stock_quantity IS NULL OR stock_quantity >= 0)
) ENGINE = InnoDB;

-- =====================================================
-- 5. PHIẾU NHẬP HÀNG
-- =====================================================
CREATE TABLE purchaseorder (
    purchase_order_id VARCHAR(10) PRIMARY KEY,
    supplier_id       VARCHAR(10),
    employee_id       VARCHAR(10),
    import_date       DATETIME,
    total_amount      DECIMAL(15, 2),
    status            VARCHAR(50),
    note              TEXT,
    CONSTRAINT fk_po_supplier FOREIGN KEY (supplier_id)
        REFERENCES supplier (supplier_id) ON DELETE RESTRICT,
    CONSTRAINT fk_po_employee FOREIGN KEY (employee_id)
        REFERENCES employee (employee_id) ON DELETE RESTRICT,
    CONSTRAINT ck_po_total CHECK (total_amount IS NULL OR total_amount >= 0)
) ENGINE = InnoDB;

-- =====================================================
-- 6. LÔ HOA
--    (đóng vai trò "chi tiết phiếu nhập" trong báo cáo)
-- =====================================================
CREATE TABLE flowerbatch (
    batch_id          VARCHAR(10) PRIMARY KEY,
    purchase_order_id VARCHAR(10),
    flower_id         VARCHAR(10),
    quantity_imported INT,
    quantity_remaining INT,
    unit_cost         DECIMAL(15, 2),
    import_date       DATETIME,
    expiry_date       DATE,
    status            VARCHAR(50),
    note              TEXT,
    CONSTRAINT fk_batch_po FOREIGN KEY (purchase_order_id)
        REFERENCES purchaseorder (purchase_order_id) ON DELETE RESTRICT,
    CONSTRAINT fk_batch_flower FOREIGN KEY (flower_id)
        REFERENCES flower (flower_id) ON DELETE CASCADE,
    CONSTRAINT ck_batch_qty CHECK (
        (quantity_imported  IS NULL OR quantity_imported  >= 0) AND
        (quantity_remaining IS NULL OR quantity_remaining >= 0)),
    CONSTRAINT ck_batch_date CHECK (
        import_date IS NULL OR expiry_date IS NULL OR DATE(import_date) <= expiry_date),
    INDEX idx_batch_flower_exp (flower_id, expiry_date)
) ENGINE = InnoDB;

-- =====================================================
-- 7. ĐƠN HÀNG
-- =====================================================
CREATE TABLE `order` (
    order_id         VARCHAR(10) PRIMARY KEY,
    customer_id      VARCHAR(10),
    employee_id      VARCHAR(10),
    order_date       DATETIME,
    delivery_date    DATETIME,
    delivery_address VARCHAR(255),
    order_status     VARCHAR(50),
    CONSTRAINT fk_order_customer FOREIGN KEY (customer_id)
        REFERENCES customer (customer_id) ON DELETE RESTRICT,
    CONSTRAINT fk_order_employee FOREIGN KEY (employee_id)
        REFERENCES employee (employee_id) ON DELETE RESTRICT,
    CONSTRAINT ck_order_date CHECK (
        order_date IS NULL OR delivery_date IS NULL OR delivery_date >= order_date),
    INDEX idx_order_status (order_status)
) ENGINE = InnoDB;

-- =====================================================
-- 8. CHI TIẾT ĐƠN HÀNG
-- =====================================================
CREATE TABLE orderdetail (
    order_id            VARCHAR(10) NOT NULL,
    flower_id           VARCHAR(10) NOT NULL,
    quantity            INT,
    unit_price_at_sale  DECIMAL(15, 2),
    PRIMARY KEY (order_id, flower_id),
    CONSTRAINT fk_detail_order FOREIGN KEY (order_id)
        REFERENCES `order` (order_id) ON DELETE CASCADE,
    CONSTRAINT fk_detail_flower FOREIGN KEY (flower_id)
        REFERENCES flower (flower_id) ON DELETE RESTRICT,
    CONSTRAINT ck_detail_qty   CHECK (quantity IS NULL OR quantity > 0),
    CONSTRAINT ck_detail_price CHECK (unit_price_at_sale IS NULL OR unit_price_at_sale > 0)
) ENGINE = InnoDB;

-- =====================================================
-- 9. THANH TOÁN
-- =====================================================
CREATE TABLE payment (
    payment_id     VARCHAR(10) PRIMARY KEY,
    order_id       VARCHAR(10),
    amount         DECIMAL(15, 2),
    payment_method VARCHAR(50),
    payment_date   DATETIME,
    payment_status VARCHAR(50),
    CONSTRAINT fk_payment_order FOREIGN KEY (order_id)
        REFERENCES `order` (order_id) ON DELETE CASCADE,
    CONSTRAINT ck_payment_amount CHECK (amount IS NULL OR amount > 0)
) ENGINE = InnoDB;

-- =====================================================
-- 10. LỊCH SỬ CHĂM SÓC
--     Mỗi lần kiểm tra = 1 dòng (nhật ký).
--     "Tình trạng hiện tại" của 1 loài hoa = dòng mới nhất của loài đó.
-- =====================================================
CREATE TABLE careschedule (
    care_id         VARCHAR(10) PRIMARY KEY,
    flower_id       VARCHAR(10),
    employee_id     VARCHAR(10),
    check_date      DATETIME,
    freshness_level VARCHAR(50),
    water_status    VARCHAR(50),
    action_taken    VARCHAR(255),
    note            TEXT,
    CONSTRAINT fk_care_flower FOREIGN KEY (flower_id)
        REFERENCES flower (flower_id) ON DELETE RESTRICT,
    CONSTRAINT fk_care_employee FOREIGN KEY (employee_id)
        REFERENCES employee (employee_id) ON DELETE RESTRICT,
    CONSTRAINT ck_care_fresh CHECK (
        freshness_level IS NULL OR
        freshness_level IN ('TƯƠI', 'HƠI HÉO', 'HÉO', 'HỎNG')),
    INDEX idx_care_flower_date (flower_id, check_date)
) ENGINE = InnoDB;

-- =====================================================
-- KIỂM TRA NHANH SAU KHI TẠO
-- Cột chenh_lech phải = 0 cho mọi dòng.
-- =====================================================
SELECT f.flower_id,
       f.name,
       f.stock_quantity AS ton_kho,
       COALESCE((SELECT SUM(fb.quantity_remaining)
                 FROM flowerbatch fb
                 WHERE fb.flower_id = f.flower_id
                   AND fb.status = 'AVAILABLE'), 0) AS tong_lo_available,
       f.stock_quantity - COALESCE((SELECT SUM(fb.quantity_remaining)
                                    FROM flowerbatch fb
                                    WHERE fb.flower_id = f.flower_id
                                      AND fb.status = 'AVAILABLE'), 0) AS chenh_lech
FROM flower f
ORDER BY f.flower_id;
